# Worker dependencies. Meant to be installed on whatever machine runs the
# transcription (your laptop, a Colab notebook, a GPU box, ...) — independent
# from the API's requirements.txt.

# Pin av explicitly: faster-whisper 1.1.0 only declares "av>=11" with no upper
# bound, and newer av releases (19.x, pulled in by default) dropped the
# metadata_errors kwarg that faster-whisper's audio decoder passes to av.open(),
# breaking every transcription with "TypeError: open() got an unexpected
# keyword argument 'metadata_errors'".
av==13.1.0
requests==2.32.3
psycopg2-binary==2.9.10
# Pinned >=2.33: older supabase-py versions reject the newer sb_secret_/sb_publishable_ API key formats.

huggingface_hub==0.36.2
